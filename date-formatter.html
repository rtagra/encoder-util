Sample code — format “now” in each language

// JavaScript
const now = new Date();
now.toISOString(); // 2026-11-11T09:40:00.000Z
new Intl.DateTimeFormat('en-GB', {dateStyle:'medium', timeStyle:'medium'}).format(now);

// Python
from datetime import datetime
datetime.now().strftime('%Y-%m-%d %H:%M:%S')  # 2026-11-11 09:40:00
datetime.now().isoformat()                   # 2026-11-11T09:40:00

// Java
import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
LocalDateTime.now().format(DateTimeFormatter.ofPattern("yyyy-MM-dd HH:mm:ss"));
LocalDateTime.now().format(DateTimeFormatter.ISO_LOCAL_DATE_TIME);

// Go
time.Now().Format("2006-01-02 15:04:05")  // Go uses a reference date, not yyyy-MM-dd
time.Now().Format(time.RFC3339)           // 2026-11-11T09:40:00+08:00

// C#
DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss");
DateTime.Now.ToString("o");  // ISO 8601

// PHP
date('Y-m-d H:i:s');  // 2026-11-11 09:40:00
date('c');            // ISO 8601
